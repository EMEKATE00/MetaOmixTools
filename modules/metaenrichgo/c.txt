# R/modules/metaenrichgo/mod_metaEnrichGO.R

mod_metaEnrichGO_server <- function(id) {
  moduleServer(id, function(input, output, session) {
    ns <- session$ns
    
    exampleFiles <- c("./modules/metaenrichgo/example_data/genes1.txt",
                      "./modules/metaenrichgo/example_data/genes2.txt",
                      "./modules/metaenrichgo/example_data/genes3.txt",
                      "./modules/metaenrichgo/example_data/genes4.txt")
    
    # --------------------------------------------
    # 1) Store uploaded or pasted input
    # --------------------------------------------
    storedInput <- reactiveVal(NULL)
    
    selectedDatabase <- eventReactive(input$runButton, {
      input$database})
    
    observe({
      req(input$inputMethod)
      
      if (input$inputMethod == "files") {
        if (isTRUE(input$useExampleData)) {
          storedInput(list(type = "files", paths = exampleFiles, names = basename(exampleFiles)))
        } else {
          req(input$files)
          storedInput(list(type = "files", paths = input$files$datapath, names = input$files$name))}
        
      } else if (input$inputMethod == "paste") {
        req(input$pastedGenes)
        storedInput(list(type = "paste", text = input$pastedGenes))}})
    
    # --------------------------------------------
    # 2) Read & clean raw gene lists with progress
    # --------------------------------------------
    rawGeneLists <- reactive({
      inputData <- storedInput()
      req(inputData)
      
      withProgress(message = "Loading gene lists...", value = 0, {
        incProgress(0.2, detail = "Reading input...")
        raw_lists <- list()
        
        if (inputData$type == "files") {
          for (i in seq_along(inputData$paths)) {
            fp <- inputData$paths[i]
            lines <- readLines(fp, warn = FALSE, encoding = "UTF-8")
            raw_lists[[ inputData$names[i] ]] <- lines}
        } else {
          blocks <- strsplit(inputData$text, "###", fixed = TRUE)[[1]]
          for (i in seq_along(blocks)) {
            raw_lists[[ paste0("List", i) ]] <- unlist(strsplit(blocks[[i]], "\\s+"))}}
        
        incProgress(0.7, detail = "Cleaning gene symbols...")
        clean_genes <- function(vec) {
          vec <- trimws(vec)
          vec <- gsub("///.*|\\\\.*", "", vec)
          vec[vec != ""]}
        
        mapping <- lapply(raw_lists, clean_genes)
        
        incProgress(1, detail = "Done.")
        mapping})})
    
    # --------------------------------------------
    # Helper: compute GeneCount column from GeneID string ("/" or whitespace separated)
    # --------------------------------------------
    compute_gene_count <- function(geneid_vec) {
      vapply(geneid_vec, function(txt) {
        if (is.na(txt) || txt == "" ) return(0L)
        genes <- unlist(strsplit(as.character(txt), "[/\\s]+"))
        genes <- genes[genes != ""]
        length(unique(genes))
      }, integer(1))}
    
    # --------------------------------------------
    # totalLists reactive
    # --------------------------------------------
    totalLists <- reactive({
      req(input$inputMethod)
      if (isTRUE(input$useExampleData)) {
        4
      } else if (input$inputMethod == "files" && !is.null(input$files)) {
        length(input$files$datapath)
      } else if (input$inputMethod == "paste" &&
                 !is.null(input$pastedGenes) && nzchar(input$pastedGenes)) {
        txt <- input$pastedGenes
        if (grepl("###", txt, fixed = TRUE)) {
          length(strsplit(txt, "###", fixed = TRUE)[[1]])
        } else {
          1}
      } else {
        0}})
    
    # --------------------------------------------
    # 3) Results reactiveVal & runContext
    # --------------------------------------------
    resultsVal <- reactiveVal(list(included = NULL, excluded = NULL))
    runContext  <- reactiveVal(NULL)   # store ORA_data + lists for per-file tabs
    
    observeEvent(input$runButton, {
      inputData <- storedInput()
      req(inputData)
      
      n_lists <- totalLists()
      lists   <- rawGeneLists()
      genes   <- unique(unlist(lists))
      
      if (input$database == "Personalized" && is.null(input$personalizedAnnotations)) {
        showModal(modalDialog(
          title = "⚠️ Annotation File Required",
          "Please upload an annotation file to perform the personalized analysis.",
          easyClose = TRUE,
          footer = modalButton("Close")))
        return()}
      
      is_personalized <- (input$database == "Personalized")
      
      if (is_personalized) {
        if (!check_annotation_file_format(input$personalizedAnnotations$datapath, session)) {
          resultsVal(list(included = NULL, excluded = NULL))
          return()}}
      
      if (is_personalized) {
        all_cols <- c("Description", "pvalue", "p.adjust", "ListCount", "ListNames", "FeatureCount", "FeatureID")
        default_selected <- if (n_lists > 1) all_cols else setdiff(all_cols, c("ListCount", "ListNames"))
        updateSelectInput(session, "plotLabel", choices = c("Description" = "Description"), selected = "Description")
      } else {
        all_cols <- c("ID", "Description", "pvalue", "p.adjust", "ListCount", "ListNames", "GeneCount", "GeneID")
        default_selected <- if (n_lists > 1) all_cols else setdiff(all_cols, c("ListCount", "ListNames"))
        updateSelectInput(session, "plotLabel", 
                          choices  = c("ID" = "ID", "Description" = "Description", "ID + Description" = "Term_Description"),
                          selected = "Term_Description")}
      
      updateCheckboxGroupInput(session, "selectedColumns",
                               choices  = all_cols,
                               selected = default_selected)
      
      # 1) Validaciones de formato
      if (inputData$type == "files" && !isTRUE(input$useExampleData)) {
        if (!validate_file_format_metaenrichgo(input, session)) {
          resultsVal(list(included = NULL, excluded = NULL))
          return()}}
      
      if (inputData$type == "paste") {
        if (!validate_paste_format(input, session)) {
          resultsVal(list(included = NULL, excluded = NULL))
          return()}}
      
      if (input$database != "Personalized") {
        if (!validate_gene_ids(genes, input$IDtype, session)) {
          resultsVal(list(included = NULL, excluded = NULL))
          return()}
        
        if (!validate_organism(genes, input$organism, input$IDtype, session)) {
          resultsVal(list(included = NULL, excluded = NULL))
          return()}}
      
      # 3) ORA por lista (siempre)
      withProgress(message = "Performing ORA...", value = 0, {
        ORA_data <- switch(input$database,
                           GO       = lapply(lists, ORA_GO,
                                             organism = input$organism,
                                             ontology = input$ontology,
                                             pvalue   = 1,
                                             ID_type  = input$IDtype),
                           KEGG     = lapply(lists, ORA_KEGG,
                                             organism = input$organism,
                                             pvalue   = 1,
                                             ID_type  = input$IDtype),
                           Reactome = lapply(lists, ORA_REACTOME,
                                             organism = input$organism,
                                             pvalue   = 1,
                                             ID_type  = input$IDtype),
                           Personalized = lapply(lists, ORA_PERSONALIZED,
                                                 annotation_file = input$personalizedAnnotations$datapath))
        
        ORA_data <- lapply(ORA_data, function(df) {
          
          if (input$database == "Personalized") {
            # Caso Personalizado (df tiene Description, FeatureID, FeatureCount)
            
            # 1. Creamos 'ID' (para el meta-análisis) a partir de 'Description'
            df$ID        <- df$Description 
            
            # 2. Creamos nombres 'Gene' estándar a partir de 'Feature'
            df$GeneID    <- df$FeatureID
            df$GeneCount <- df$FeatureCount
            
            # 3. ¡IMPORTANTE! Eliminamos los 'Feature' originales
            #    para evitar duplicados más adelante.
            df$FeatureID   <- NULL
            df$FeatureCount <- NULL
            
          } else {
            # Caso GO/KEGG/Reactome (formato estándar)
            if ("geneID" %in% colnames(df) && !("GeneID" %in% colnames(df)))
              colnames(df)[colnames(df) == "geneID"] <- "GeneID"
            if (!("GeneID" %in% colnames(df))) df$GeneID <- ""
            if (!("GeneCount" %in% colnames(df))) 
              df$GeneCount <- compute_gene_count(df$GeneID)
          }
          
          # Asegurar columnas comunes
          if (!("pvalue" %in% colnames(df))) df$pvalue <- NA_real_
          if (!("Description" %in% colnames(df))) df$Description <- NA_character_
          if (!("ID" %in% colnames(df))) df$ID <- NA_character_
          
          # Añadir p.adjust si falta
          if (!("p.adjust" %in% colnames(df))) 
            df$p.adjust <- p.adjust(df$pvalue, method = "BH")
          
          return(df)
        })
        incProgress(1, detail = "ORA Completed")})
      
      if (is.null(names(lists)) || any(names(lists) == "")) {
        names(ORA_data) <- paste0("List", seq_along(ORA_data))
      } else {
        names(ORA_data) <- names(lists)}
      runContext(list(ORA_data = ORA_data, lists = lists))
      
      # 4) Si hay >1 lista: meta-análisis → generar tabla combinada
      if (n_lists > 1) {
        withProgress(message = "Performing Meta-Analysis...", value = 0, {
          termLists     <- lapply(ORA_data, `[[`, "ID")
          termFrequency <- table(unlist(termLists))
          totalFiles    <- n_lists
          minFreq       <- input$termThreshold
          commonTerms   <- names(termFrequency)[termFrequency >= minFreq]
          totalTerms    <- length(commonTerms)
          
          METAresults <- data.frame(
            ID = character(0), Description = character(0),
            pvalue = numeric(0), p.adjust = numeric(0),
            ListCount = character(0), ListNames = character(0),
            GeneCount = integer(0), GeneID = character(0),
            stringsAsFactors = FALSE)
          
          for (i in seq_along(commonTerms)) {
            term <- commonTerms[i]
            
            pvs <- na.omit(sapply(ORA_data, function(res) {
              td <- res[res$ID == term, ]
              if (nrow(td)) td$pvalue else NA}))
            
            combP <- switch(input$metaAnalysisMethod,
                            fisher    = fisher_combined_pvalue(pvs),
                            stouffer  = stouffer_combined_pvalue(pvs),
                            tippett   = tippett_combined_pvalue(pvs),
                            wilkinson = wilkinson_combined_pvalue(pvs))
            
            desc <- NA_character_
            for (res in ORA_data) {
              td <- res[res$ID == term, ]
              if (nrow(td)) { desc <- td$Description[1]; break }}
            
            genesIn <- unique(unlist(lapply(ORA_data, function(res) {
              td <- res[res$ID == term, ]
              if (nrow(td)) strsplit(as.character(td$GeneID), "[/\\s]+")[[1]] else character(0)})))
            genesIn <- genesIn[genesIn != ""]
            geneStr <- paste(genesIn, collapse = "/ ")
            filesWith <- names(ORA_data)[sapply(ORA_data, function(res) term %in% res$ID)]
            
            METAresults <- rbind(METAresults, data.frame(
              ID = term,
              Description = desc,
              pvalue = combP,
              p.adjust = NA_real_,
              ListCount = paste0(length(filesWith), "/", totalFiles),
              ListNames = paste(filesWith, collapse = ", "),
              GeneCount = length(genesIn),
              GeneID = geneStr,
              stringsAsFactors = FALSE))
            
            incProgress(1 / max(totalTerms, 1), detail = paste("Term", i, "of", totalTerms))}
          
          METAresults$p.adjust <- p.adjust(METAresults$pvalue, method = "BH")
          METAresults <- METAresults[order(METAresults$p.adjust), ]
          
          allT <- names(termFrequency)
          lowT <- setdiff(allT, commonTerms)
          EXCres <- data.frame(
            ID = character(0), Description = character(0),
            ListCount = character(0), ListNames = character(0),
            GeneCount = integer(0), GeneID = character(0),
            stringsAsFactors = FALSE)
          
          for (term in lowT) {
            desc <- NA_character_
            for (res in ORA_data) {
              td <- res[res$ID == term, ]
              if (nrow(td)) { desc <- td$Description[1]; break }}
            genesIn <- unique(unlist(lapply(ORA_data, function(res) {
              td <- res[res$ID == term, ]
              if (nrow(td)) strsplit(as.character(td$GeneID), "[/\\s]+")[[1]] else character(0)
            })))
            genesIn <- genesIn[genesIn != ""]
            filesWith <- names(ORA_data)[sapply(ORA_data, function(res) term %in% res$ID)]
            EXCres <- rbind(EXCres, data.frame(
              ID = term,
              Description = desc,
              ListCount = paste0(length(filesWith), "/", totalFiles),
              ListNames = paste(filesWith, collapse = ", "),
              GeneCount = length(genesIn),
              GeneID = paste(genesIn, collapse = "/ "),
              stringsAsFactors = FALSE))}
          
          if (is_personalized) {
            colnames(METAresults)[colnames(METAresults) == "GeneCount"] <- "FeatureCount"
            colnames(METAresults)[colnames(METAresults) == "GeneID"] <- "FeatureID"
            METAresults$ID <- NULL
            
            colnames(EXCres)[colnames(EXCres) == "GeneCount"] <- "FeatureCount"
            colnames(EXCres)[colnames(EXCres) == "GeneID"] <- "FeatureID"
            EXCres$ID <- NULL
          }
          
          if (!check_empty_results(METAresults, session)) {
            resultsVal(list(included = NULL, excluded = NULL))
            return()}
          
          resultsVal(list(included = METAresults, excluded = EXCres))})
      } else { 
        
        df <- ORA_data[[1]]
        if ("geneID" %in% colnames(df) && !("GeneID" %in% colnames(df)))
          colnames(df)[colnames(df) == "geneID"] <- "GeneID"
        if (!("p.adjust" %in% colnames(df))) df$p.adjust <- p.adjust(df$pvalue, method = "BH")
        if (!("GeneCount" %in% colnames(df))) df$GeneCount <- compute_gene_count(df$GeneID)
        
        list_name <- if (!is.null(names(lists)) && length(names(lists)) >= 1) names(lists)[1] else "List1"
        df$ListCount <- "1/1"
        df$ListNames <- list_name
        
        if (is_personalized) {
          # df (que viene de ORA_list) SÓLO tiene GeneCount y GeneID.
          # Los renombramos a FeatureCount y FeatureID.
          # Como los 'Feature' originales fueron eliminados en el Paso 2,
          # no se crea ningún duplicado.
          colnames(df)[colnames(df) == "GeneCount"] <- "FeatureCount"
          colnames(df)[colnames(df) == "GeneID"]    <- "FeatureID"
          df$ID <- NULL

    #' Validate files for MetaEnrichGO
    #' @param input Shiny input object
    #' @param session Shiny session object
    #' @return TRUE if valid; FALSE if any file had wrong format
    
    validate_file_format_metaenrichgo <- function(input, session) {
      bad_files <- list()
      
      files <- if (input$inputMethod == "files") input$files$datapath else NULL
      file_names <- if (input$inputMethod == "files") input$files$name else NULL
      
      if (is.null(files)) return(TRUE)
      
      for (i in seq_along(files)) {
        lines <- readLines(files[i], warn = FALSE, encoding = "UTF-8")
        lines <- trimws(lines[nzchar(lines)]) 
        
        if (any(grepl("#", lines))) {
          bad_files[[file_names[i]]] <- "Contains invalid characters like '#'."
          next}
        
        if (any(grepl("[,\t]", lines))) {
          bad_files[[file_names[i]]] <- "Files must contain only one gene symbol per line (no commas or tabs)."
          next}}
      
      if (length(bad_files) > 0) {
        showModal(modalDialog(
          title = "⚠️ Format Errors Detected",
          size = "l",
          easyClose = TRUE,
          footer = modalButton("Close"),
          tags$ul(
            lapply(names(bad_files), function(f) {
              tags$li(HTML(paste("<b>", f, "</b>: ", bad_files[[f]])))
            })
          ),
          tags$hr(),
          fluidRow(
            column(12,
                   tags$h4("Expected Format"),
                   tags$p("Each uploaded file must contain a plain list of gene identifiers, one per line, with no headers or separators."),
                   tags$h5("Valid Examples by ID Type and Organism"),
                   fluidRow(
                     column(4, wellPanel(
                       tags$strong("SYMBOL / HUMAN"),
                       HTML("<pre>BRCA1\nTP53\nMYC</pre>"))),
                     column(4, wellPanel(
                       tags$strong("SYMBOL / MOUSE"),
                       HTML("<pre>Trp53\nMyc\nEgfr</pre>"))),
                     column(4, wellPanel(
                       tags$strong("SYMBOL / RAT"),
                       HTML("<pre>Tp53\nMyc\nEgfr</pre>")))),
                   fluidRow(
                     column(4, wellPanel(
                       tags$strong("ENTREZID / HUMAN"),
                       HTML("<pre>672\n7157\n4609</pre>"))),
                     column(4, wellPanel(
                       tags$strong("ENTREZID / MOUSE"),
                       HTML("<pre>22059\n17869\n13649</pre>"))),
                     column(4, wellPanel(
                       tags$strong("ENTREZID / RAT"),
                       HTML("<pre>24842\n116538\n25499</pre>")))),
                   fluidRow(
                     column(4, wellPanel(
                       tags$strong("ENSEMBL / HUMAN"),
                       HTML("<pre>ENSG00000012048\nENSG00000141510\nENSG00000136997</pre>"))),
                     column(4, wellPanel(
                       tags$strong("ENSEMBL / MOUSE"),
                       HTML("<pre>ENSMUSG00000059552\nENSMUSG00000059552\nENSMUSG00000020122</pre>"))),
                     column(4, wellPanel(
                       tags$strong("ENSEMBL / RAT"),
                       HTML("<pre>ENSRNOG00000016647\nENSRNOG00000019902\nENSRNOG00000000007</pre>"))))))))
        return(FALSE)}
      return(TRUE)}

    #' Validate organism nomenclature
    #' @param geneLists Character vector of genes to validate
    #' @param organism  One of "Hsa", "Mmu", "Rno"
    #' @param session   Shiny session (para showModal)
    #' @return TRUE if valid; FALSE (and shows modal) if any invalid genes found
    
    validate_organism <- function(gene_list, organism, ID_type, session) {
      if (length(gene_list) == 0) return(TRUE)
      
      # Define pattern based on ID_type and organism
      pattern <- switch(ID_type,
                        SYMBOL = switch(organism,
                                        Hsa = "^[A-Z][A-Z0-9._/\\-]*$",         # Human SYMBOL
                                        Mmu = "^[A-Z][a-z0-9._/\\-]*$",         # Mouse SYMBOL
                                        Rno = "^[A-Z][a-z0-9._/\\-]*$",         # Rat SYMBOL
                                        NULL),
                        ENTREZID = "^[0-9]+$",                                  # Only digits
                        ENSEMBL = switch(organism,
                                         Hsa = "^ENSG[0-9]{11}$",               # Human ENSEMBL
                                         Mmu = "^ENSMUSG[0-9]{11}$",            # Mouse ENSEMBL
                                         Rno = "^ENSRNOG[0-9]{11}$",            # Rat ENSEMBL

    #' Validate geneid format

    #' @param session   Shiny session (para showModal)
    #' @return TRUE if valid; FALSE (and shows modal) if any invalid genes found
    
    validate_gene_ids <- function(gene_vector, ID_type, session) {
      if (length(gene_vector) == 0) return(TRUE)
      
      gene_list <- as.character(gene_vector)
      
      ensembl_pattern <- "^ENS[A-Z]{0,4}[0-9]{6,}$"
      pattern <- switch(ID_type,
                        SYMBOL   = "^[A-Za-z][A-Za-z0-9._/\\-]*$",
                        ENTREZID = "^[0-9]+$",
                        ENSEMBL  = ensembl_pattern,
                        stop("Unsupported ID type"))
      
      is_valid <- grepl(pattern, gene_list)

      return(TRUE)}
    
    
    #' Check if enrichment results are empty and show modal if so
    #' @param results  Data.frame or object with enrichment results
    #' @param session  Shiny session object
    #' @return TRUE if results are non-empty; FALSE (and shows modal) if empty
    
    check_empty_results <- function(results, session) {
      is_empty <- is.null(results) || (is.data.frame(results) && nrow(results) == 0)
      
      if (is_empty) {
        showModal(modalDialog(
          title = tags$div("⚠️ No Enrichment Results Found", 
                           style = "text-align: center; font-weight: bold; font-size: 20px;"),
          easyClose = TRUE,
          size = "l",
          tags$h4("No biological terms matched your input genes"),
          tags$p("The enrichment analysis did not find any relevant pathways, terms or functions associated with the provided gene list."),
          tags$hr(),
          tags$h4("Suggestions to improve your input"),
          tags$ul(
            tags$li("Check the organism selection (e.g., Human, Mouse or Rat)."),
            tags$li("Verify gene identifiers and formats (e.g., SYMBOL, ENTREZID or ENSEMBL)."),
            tags$li("Ensure your genes are annotated in the selected database."),
            tags$li("Use more genes if possible — small lists often yield no results.")),
          tags$p("If the problem persists, try example data to confirm the tool is working."),
          footer = modalButton("Close")))
        return(FALSE)}
      return(TRUE)}
    
    
    #-------------------------------------------------------------------------------
    # Personalized Annotation File Format Validation
    #-------------------------------------------------------------------------------
    check_annotation_file_format <- function(file_path, session) {
      # 1. Attempt to read the file, assuming common delimiters.
      # Use tryCatch to handle errors if the file doesn't exist or is corrupt.
      tryCatch({
        
        # Try reading with tab separator first (assuming TSV/TXT)
        df <- data.table::fread(file_path, header = TRUE, data.table = FALSE, check.names = FALSE, sep = "\t", nrows = 5)
        
        # If reading with tab failed (e.g., only 1 column), try with comma (CSV)
        if (ncol(df) <= 1) {
          df <- data.table::fread(file_path, header = TRUE, data.table = FALSE, check.names = FALSE, sep = ",", nrows = 5)
        }
        
        # 2. Check the column count
        if (ncol(df) != 2) {
          showModal(modalDialog(
            title = HTML("❌ Annotation File Format Error"),
            tags$h4(paste0("The annotation file must contain **exactly 2 columns**, but ", ncol(df), " were detected.")),
            tags$p("Ensure your file (CSV, TSV, or TXT) has only two fields: **Feature ID** and **Annotation Value/Description**, and that the correct separator was used."),
            tags$hr(),
            tags$h5("Format Suggestions:"),
            tags$ul(
              tags$li("It must include a **header row**."),
              tags$li("Check if the separator is a comma (`,`) or a tab (`\t`)."),
              tags$li("Ensure the file does not contain spurious lines or multiple consecutive delimiters.")
            ),
            easyClose = TRUE,
            footer = modalButton("Close")
          ))
          return(FALSE)
        }
        
        # If the column check passes:
        return(TRUE)
        
      }, error = function(e) {
        showModal(modalDialog(
          title = HTML("❌ File Reading Error"),
          tags$h4("The personalized annotation file could not be read."),
          tags$p(paste0("The error detected is: ", e$message)),
          tags$p("Please verify that the file is in a valid plain text format (CSV, TSV, TXT)."),
          easyClose = TRUE,
          footer = modalButton("Close")
        ))
        return(FALSE)
      })
    }
